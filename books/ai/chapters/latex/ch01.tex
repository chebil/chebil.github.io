\begin{frame}{Deductive vs. Inductive Reasoning}{The two schools of thought}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Deductive Reasoning}
        \small Start from a \keyword{knowledge base} of facts; apply logic to make
        inferences.
        \[ \text{Rules} + \text{Facts} \Rightarrow \text{Conclusions} \]
        \textit{Search and logic-based methods.}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{Inductive Reasoning}
        \small Start from \keyword{observations}; generalize to broader rules or
        patterns.
        \[ \text{Examples} \Rightarrow \text{General Rule} \]
        \textit{The foundation of machine learning.}
      \end{exampleblock}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \begin{center}\small
    \important{The greatest strength of deductive reasoning — interpretability —
    is also its greatest limitation.}
  \end{center}
\end{frame}

\begin{frame}{Deductive Methods}{Systematic inference from knowledge}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \keyword{Constraint Satisfaction (CSP)}\\
          \scriptsize variables, domains, constraints
        \item \keyword{Search Algorithms}\\
          \scriptsize BFS, DFS, UCS, A$^\ast$, local search
        \item \keyword{Logic-Based Systems}\\
          \scriptsize propositional \& first-order logic, resolution
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Expert Systems}
        \small
        \begin{itemize}
          \item \keyword{Knowledge base}: facts + rules
          \item \keyword{Inference engine}: applies rules
          \item \keyword{User interface}
        \end{itemize}
        \scriptsize Forward chaining (data-driven) \textbullet\
        Backward chaining (goal-driven)
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Constraint Satisfaction Problems}
% ============================================================

\begin{frame}{Constraint Satisfaction Problems (CSPs)}{A deductive modeling framework}
  \begin{columns}[c]
    \begin{column}{0.58\textwidth}
      A \keyword{CSP} is defined by three ingredients:
      \begin{itemize}\small
        \item \keyword{Variables} $X_1,\dots,X_n$
        \item \keyword{Domains} — allowed values for each variable
        \item \keyword{Constraints} — legal combinations of values
      \end{itemize}
      \vspace{0.3em}
      A \emph{solution} assigns a value to every variable so that \important{all
      constraints hold}.
      \vspace{0.4em}
      \begin{exampleblock}{Everyday CSPs}
        \small Sudoku \textbullet\ $N$-Queens \textbullet\ Traveling Salesperson
        \textbullet\ map / graph coloring \textbullet\ scheduling.
      \end{exampleblock}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{center}
      \begin{tikzpicture}[node distance=0.45cm, b/.style={processstep, minimum width=3.1cm}]
        \node[b, fill=PrimaryLight!45] (v) {Variables};
        \node[b, below=of v, fill=SuccessGreen!20] (d) {Domains};
        \node[b, below=of d, fill=AccentYellow!35] (c) {Constraints};
        \node[b, below=of c, fill=AccentOrange!20, draw=AccentOrange] (s) {Solution};
        \draw[-{Latex[length=2mm]},PrimaryMid] (v)--(d);
        \draw[-{Latex[length=2mm]},PrimaryMid] (d)--(c);
        \draw[-{Latex[length=2mm]},PrimaryMid] (c)--(s);
      \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Example 1 — Sudoku as a CSP}{Fill the $9\times9$ grid}
  \begin{columns}[T]
    \begin{column}{0.6\textwidth}
      \begin{itemize}\small
        \item \keyword{Variables}: $X_{i,j}$ for each cell $(i,j)$
        \item \keyword{Domains}: $\{1,2,\dots,9\}$
        \item \keyword{Constraints} — all different within each unit:
      \end{itemize}
      \small
      \begin{block}{Row / column / sub-grid}
        \footnotesize
        \textbf{Rows:} $\forall i,\ j\neq k:\ X_{i,j}\neq X_{i,k}$ \\[0.2em]
        \textbf{Columns:} $\forall j,\ i\neq k:\ X_{i,j}\neq X_{k,j}$ \\[0.2em]
        \textbf{Sub-grids:} within each $3\times3$ block, \\
        \quad $X_{3m+i,\,3n+j}\neq X_{3m+k,\,3n+l}$ for $(i,j)\neq(k,l)$
      \end{block}
    \end{column}
    \begin{column}{0.4\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=0.38]
        \fill[PrimaryLight!30] (0,0) rectangle (9,9);
        \draw[step=1, PrimaryMid, thin] (0,0) grid (9,9);
        \draw[step=3, PrimaryDark, line width=1.2pt] (0,0) grid (9,9);
        \draw[PrimaryDark, line width=1.2pt] (0,0) rectangle (9,9);
      \end{tikzpicture}\\[0.3em]
      \scriptsize 81 variables, 27 \emph{all-different} constraints.
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Sudoku — Python Implementation}{Using the \texttt{python-constraint} library}
\begin{lstlisting}[language=Python, basicstyle=\ttfamily\scriptsize, numbers=none,
                   xleftmargin=0.5em, framexleftmargin=0.5em]
from constraint import Problem, AllDifferentConstraint

def solve_sudoku(puzzle):            # puzzle: 9x9 list, 0 = blank
    p = Problem()
    for r in range(9):
        for c in range(9):
            v = puzzle[r][c]
            p.addVariable((r, c), range(1, 10) if v == 0 else [v])
    for i in range(9):               # rows and columns all-different
        p.addConstraint(AllDifferentConstraint(), [(i, c) for c in range(9)])
        p.addConstraint(AllDifferentConstraint(), [(r, i) for r in range(9)])
    for br in range(3):              # 3x3 sub-grids all-different
        for bc in range(3):
            cells = [(br*3 + i, bc*3 + j) for i in range(3) for j in range(3)]
            p.addConstraint(AllDifferentConstraint(), cells)
    sols = p.getSolutions()
    return sols[0] if sols else None
\end{lstlisting}
\end{frame}

\begin{frame}{Example 2 — Eight Queens as a CSP}{No two queens attack each other}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \begin{itemize}\small
        \item \keyword{Variables}: $Q_1,\dots,Q_8$ — column of the queen in row $i$

        \small
        \textbf{Columns differ:} $\ \forall i\neq j:\ Q_i \neq Q_j$ \\[0.3em]

\begin{frame}{Example 3 — Traveling Salesperson as a CSP}{Shortest tour of all cities}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      Cities $\{1,\dots,n\}$ with distances $d(i,j)$; find a tour of cost at most $C$.
      \begin{itemize}\small
        \item \keyword{Variables}: $X_{i,j}\in\{0,1\}$ — $1$ if the tour goes directly $i\to j$
        \item \keyword{Domains}: $\{0,1\}$
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Constraints}
        \footnotesize
        Enter each city once: $\displaystyle\sum_{i\neq j} X_{i,j}=1$ \\[0.4em]
        Leave each city once: $\displaystyle\sum_{j\neq i} X_{i,j}=1$ \\[0.4em]
        Bounded cost: $\displaystyle\sum_{i}\sum_{j\neq i} d(i,j)\,X_{i,j}\le C$
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.3em}
  \begin{center}\scriptsize
    A tour with $n$ cities has $(n-1)!$ possibilities — \important{NP-hard} as $n$ grows.
  \end{center}
\end{frame}

\begin{frame}{Example 4 — Graph / Map Coloring as a CSP}{Adjacent regions differ}
  \begin{columns}[T]
    \begin{column}{0.6\textwidth}

      \begin{itemize}\small
        \item \keyword{Variables}: $X_{i,c}\in\{0,1\}$ — $1$ if vertex $i$ has color $c$
        \item \keyword{Domains}: $\{0,1\}$
      \end{itemize}
      \begin{block}{Constraints}
        \footnotesize

        Adjacent differ: $\ X_{i,c}+X_{j,c}\le 1,\ \forall (i,j)\in E,\ \forall c$
      \end{block}
    \end{column}
    \begin{column}{0.4\textwidth}
      \begin{center}

\begin{frame}{The Agent Abstraction}{Perceive — Decide — Act}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      An \keyword{agent}:
      \begin{enumerate}
        \item \keyword{Perceives} the environment via \emph{sensors}
        \item \keyword{Acts} on it via \emph{actuators}
        \item Makes \keyword{decisions} to achieve goals
      \end{enumerate}
      \vspace{0.4em}
      \begin{block}{Formal definition}
        \small An agent function maps percept histories to actions:
        \[ f : P^{\ast} \rightarrow A \]
      \end{block}
    \end{column}
    \begin{column}{0.52\textwidth}
      \begin{center}
      \begin{tikzpicture}[node distance=0.9cm,
          ag/.style={draw=AccentOrange, fill=AccentYellow!30, rounded corners=3pt,
            minimum width=2.6cm, minimum height=1.0cm, font=\bfseries\small, align=center},
          env/.style={draw=PrimaryMid, fill=PrimaryLight!45, rounded corners=3pt,
            minimum width=2.6cm, minimum height=1.0cm, font=\bfseries\small, align=center}]
        \node[ag] (a) {Agent\\Function};

\begin{frame}{Supervised Learning}{Learning with a teacher}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      Learn a function from \keyword{labeled} examples.
      \begin{itemize}\small
        \item Training data: $(X_1,y_1),\ldots,(X_n,y_n)$
        \item Learn $f$ such that $y_i \approx f(X_i)$
        \item Minimize loss: $\displaystyle L = \sum_{i=1}^{n}(y_i - f(X_i))^2$
      \end{itemize}
      \vspace{0.3em}
      \textbf{Two tasks}
      \begin{itemize}\small
        \item \keyword{Classification}: categorical (spam/not spam)
        \item \keyword{Regression}: numerical (house prices)
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Example}
        \small Data $(1,3),(2,5),(3,7),(4,9)$
        \[ y = 2x + 1 \]
        Predict $x=5 \Rightarrow y = 11$.
      \end{block}
      \begin{center}
      \begin{tikzpicture}
        \begin{axis}[width=5.2cm, height=3.4cm, tick style={draw=none},
          xtick=\empty, ytick=\empty, axis lines=left,
          xlabel=\scriptsize $x$, ylabel=\scriptsize $y$, xmin=0, xmax=5.5, ymin=0, ymax=12]
          \addplot[only marks, mark=*, PrimaryMid] coordinates {(1,3)(2,5)(3,7)(4,9)};
          \addplot[AccentOrange, thick, domain=0.5:5] {2*x+1};
        \end{axis}
      \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Unsupervised Learning}{Discovering hidden structure}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      Find structure in \keyword{unlabeled} data.
      \begin{itemize}\small
        \item \keyword{Clustering}: K-means, hierarchical, DBSCAN
        \item \keyword{Dimensionality reduction}: PCA, t-SNE
        \item \keyword{Anomaly detection}: fraud, intrusion
      \end{itemize}
      \vspace{0.3em}
      \begin{block}{K-Means (sketch)}
        \scriptsize
        1. Initialize $k$ random centroids \\
        2. Repeat until convergence: \\
        \quad a. Assign each point to nearest centroid \\
        \quad b. Update centroids to cluster means
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{center}
      \begin{tikzpicture}
        \begin{axis}[width=5.4cm, height=4.2cm, tick style={draw=none},

      \scriptsize Three clusters discovered
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Reinforcement Learning}{Learning from interaction}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      An \keyword{agent} learns which actions to take by observing \keyword{rewards}.
      \begin{itemize}\small
        \item State $s$, action $a$, reward $r$, policy $\pi(s)$
        \item Goal: maximize cumulative reward
      \end{itemize}
      \[ G_t = \sum_{k=0}^{\infty}\gamma^k r_{t+k},\quad \gamma\in[0,1] \]
      \scriptsize Algorithms: Q-Learning, DQN (value-based); Policy Gradient,
      Actor–Critic (policy-based); AlphaZero (model-based).
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{center}
      \begin{tikzpicture}[node distance=1.7cm,
          b/.style={draw, rounded corners=3pt, minimum width=2.4cm,

        \node[b, fill=AccentYellow!30, draw=AccentOrange] (a) {Agent};
        \node[b, fill=PrimaryLight!45, draw=PrimaryMid, right=of a] (e) {Environment};
        \draw[-{Latex[length=2mm]}, AccentOrange, thick]
          (a.north) to[out=45,in=135] node[above, font=\scriptsize] {action $a_t$} (e.north);
        \draw[-{Latex[length=2mm]}, PrimaryMid, thick]
          (e.south) to[out=-135,in=-45] node[below, font=\scriptsize] {state $s_{t+1}$, reward $r_t$} (a.south);
      \end{tikzpicture}
      \end{center}
      \vspace{0.3em}
      \begin{exampleblock}{Applications}
        \scriptsize AlphaGo, robotics, autonomous vehicles, data-center cooling.
      \end{exampleblock}

  \end{columns}
