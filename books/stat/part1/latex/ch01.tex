\section{What Is a Dataset?}
% ============================================================

\begin{frame}{What Is a Dataset?}{A collection of measurements, described as $d$-tuples}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{Definition}
        \small A \keyword{dataset} $\{x\}$ is a collection of $N$ items;
        each item is a \keyword{$d$-tuple} of measurements,
        $x_i = (x_i^1, x_i^2, \dots, x_i^d)$.
      \end{block}
      \begin{itemize}\small
        \item $N$ — number of items (rows)
        \item $d$ — number of attributes (columns)
        \item Each attribute is a \keyword{dimension} of the tuple
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Students dataset ($d=3$)}
        \small
        \renewcommand{\arraystretch}{1.2}
        \begin{tabular}{@{}lcc@{}}
          \toprule
          \textbf{Name} & \textbf{Age} & \textbf{GPA} \\
          \midrule
          Alice & 21 & 3.7 \\
          Bob   & 23 & 3.2 \\
          Cara  & 22 & 3.9 \\
          \bottomrule
        \end{tabular}
      \end{exampleblock}
      \centering\scriptsize $N = 3$ items, $d = 3$ attributes each
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Types of Data}{Choosing the right tool starts with the right type}
  \small
  \renewcommand{\arraystretch}{1.3}
  \begin{center}
  \begin{tabularx}{\textwidth}{@{}>{\bfseries}l >{\raggedright\arraybackslash}X >{\raggedright\arraybackslash}X@{}}
    \toprule
    \rowcolor{PrimaryDark}
    \textcolor{white}{Type} & \textcolor{white}{Meaning} & \textcolor{white}{Examples} \\
    \midrule
    Categorical & Labels, no order & Country, blood type, gender \\
    \rowcolor{PrimaryLight!25}
    Ordinal & Labels, natural order & Letter grade, satisfaction (low/med/high) \\
    Discrete & Countable numeric values & \# of siblings, \# of goals scored \\
    \rowcolor{PrimaryLight!25}
    Continuous & Any value in a range & Height, temperature, time \\
    \bottomrule

  \begin{alertblock}{Why it matters}
    \small The type of data determines which summaries and plots make sense —
    a mean of postal codes is meaningless, but a mean of temperatures is not.
  \end{alertblock}

  \begin{columns}[T]

        \item \keyword{Complement} $\{x\}^c$ — relative to universe $\Omega$

\begin{frame}{Histograms}{The workhorse plot for numeric data}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \begin{block}{Definition}
        \small Split the range into \keyword{bins}; bar height $=$ count of
        values falling in that bin.
      \end{block}
      \begin{alertblock}{Choosing bin count}
        \small
        \begin{itemize}
          \item Too few bins $\Rightarrow$ hides structure
          \item Too many bins $\Rightarrow$ noisy, sparse
          \item Rule of thumb: $\sqrt{N}$ or Sturges' rule
        \end{itemize}

    \end{column}

\section{Measures of Central Tendency}
% ============================================================

\begin{frame}{The Mean}{The balance point of the data}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Definition}
        \[ \text{mean}(\{x\}) = \bar{x} = \frac{1}{N}\sum_{i=1}^{N} x_i \]
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{Properties}
        \small
        \begin{itemize}
          \item Translation: $\text{mean}(\{x+c\}) = \bar{x}+c$
          \item Scaling: $\text{mean}(\{kx\}) = k\bar{x}$
          \item Minimizes $\sum_i (x_i - v)^2$ over $v$
          \item \important{Sensitive} to outliers
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Computing Mean \& Std.\ Dev.\ Online}{Update estimates as data streams in}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      After seeing $k$ items, let $\hat\mu_k$, $\hat\sigma_k$ be the running
      estimates. On seeing $x_{k+1}$:
      \begin{block}{Mean update}
        \[ \hat\mu_{k+1} = \frac{k\,\hat\mu_k + x_{k+1}}{k+1} \]
      \end{block}
      \begin{block}{Std.\ dev.\ update}
        \[ \hat\sigma_{k+1} = \sqrt{\frac{k\,\hat\sigma_k^2 + (x_{k+1}-\hat\mu_{k+1})^2}{k+1}} \]
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Why it matters}
        \small
        \begin{itemize}
          \item No need to store all $N$ values
          \item Constant memory, constant work per item
          \item Ideal for \keyword{streaming} sensor / log data
        \end{itemize}
      \end{exampleblock}
      \centering\scriptsize $\mathcal{O}(1)$ time and space per update
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The Median}{The middle value, robust to outliers}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Definition}
        \small Sort $\{x\}$; the median is the middle value
        ($N$ odd) or the average of the two middle values ($N$ even).
      \end{block}
      \begin{itemize}\small
        \item \keyword{Robust}: unaffected by extreme outliers
        \item Splits data into two equal halves
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{Median vs.\ mean with outliers}
        \small Salaries: $\{30, 32, 35, 33, 500\}$ (in \$k)
        \begin{align*}
          \text{mean} &= 126 \\
          \text{median} &= 33
        \end{align*}
        \small The median better reflects a \keyword{typical} salary.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Measures of Spread}
% ============================================================

\begin{frame}{Variance \& Standard Deviation}{How far, on average, from the mean?}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Definition}
        \[ \text{var}(\{x\}) = \sigma^2 = \frac{1}{N}\sum_{i=1}^N (x_i - \bar x)^2 \]
        \[ \text{std}(\{x\}) = \sigma = \sqrt{\text{var}(\{x\})} \]
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{Properties}
        \small
        \begin{itemize}
          \item Translation invariant: $\text{std}(\{x+c\})=\sigma$
          \item Scaling: $\text{std}(\{kx\}) = |k|\,\sigma$
          \item Same \keyword{units} as the data (unlike variance)
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Percentiles, Quartiles \& the IQR}{Describing spread without assuming a shape}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Percentiles \& quartiles}
        \small
        \begin{itemize}
          \item $\text{percentile}(\{x\},k)$: value below which $k\%$ of data lie
          \item $Q1 = \text{percentile}(\{x\},25)$
          \item $Q2 = \text{percentile}(\{x\},50) = $ median
          \item $Q3 = \text{percentile}(\{x\},75)$
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{Interquartile range}
        \[ \text{IQR}(\{x\}) = Q3 - Q1 \]
        \small Robust measure of the spread of the \keyword{middle 50\%}.
        \begin{itemize}\small
          \item Translation invariant: $\text{IQR}(\{x+c\})=\text{IQR}(\{x\})$
          \item Scaling: $\text{IQR}(\{kx\}) = |k|\,\text{IQR}(\{x\})$
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Detecting Outliers with the IQR}{The rule behind every box-plot whisker}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \begin{alertblock}{The $1.5\times$IQR rule}
        \small An item $x_i$ is an outlier if
        \[ x_i < Q1 - 1.5\,\text{IQR} \]
        \[ \text{or} \quad x_i > Q3 + 1.5\,\text{IQR} \]
      \end{alertblock}
    \end{column}
    \begin{column}{0.55\textwidth}
      \begin{center}
      \resizebox{\linewidth}{!}{%
      \begin{tikzpicture}[x=1.05cm]
        \draw[-{Latex[length=2mm]}, PrimaryDark, thick] (0,0) -- (9.5,0);
        \draw[AlertRed, thick] (0.3,-0.15) -- (0.3,0.15);
        \draw[PrimaryMid, thick] (1.8,-0.2) -- (1.8,0.2)
          node[above, font=\scriptsize] {$Q1{-}1.5\text{IQR}$};
        \fill[PrimaryLight!55] (3,-0.35) rectangle (6,0.35);
        \draw[PrimaryDark, thick] (4.5,-0.35) -- (4.5,0.35);
        \draw[PrimaryMid, thick] (7.2,-0.2) -- (7.2,0.2)
          node[above, font=\scriptsize] {$Q3{+}1.5\text{IQR}$};
        \draw[AlertRed, thick] (9,-0.15) -- (9,0.15);
        \node[AlertRed, font=\scriptsize] at (0.3,-0.5) {outlier};
        \node[AlertRed, font=\scriptsize] at (9,-0.5) {outlier};
        \node[font=\scriptsize] at (4.5,0.65) {median};
      \end{tikzpicture}}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\section{Standard Coordinates \& Box Plots}
% ============================================================

\begin{frame}{Standard Coordinates (Z-Scores)}{Comparing values across different scales}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Definition}
        \[ \hat{x}_i = \frac{x_i - \bar{x}}{\sigma} \]
      \end{block}
      \begin{itemize}\small
        \item $\text{mean}(\{\hat x\}) = 0$ \emph{always}
        \item $\text{std}(\{\hat x\}) = 1$ \emph{always}
        \item \keyword{Unitless} — comparable across measurements
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{Reading a z-score}
        \small
        \renewcommand{\arraystretch}{1.25}
        \begin{tabular}{@{}cl@{}}
          \toprule
          $\hat x$ & Interpretation \\
          \midrule
          $0$ & exactly at the mean \\
          $+1$ & 1 s.d.\ above mean \\
          $-2$ & 2 s.d.\ below mean \\
          \bottomrule
        \end{tabular}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Anatomy of a Box Plot}{Five numbers, one picture}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{center}
      \begin{tikzpicture}[y=1cm]

        \node[right, font=\scriptsize] at (0.55,3.6) {max (within $1.5$IQR)};
        \node[right, font=\scriptsize] at (0.55,2.4) {$Q3$};
        \node[right, font=\scriptsize] at (0.55,1.9) {median};
        \node[right, font=\scriptsize] at (0.55,1.4) {$Q1$};
        \node[right, font=\scriptsize] at (0.55,0.4) {min (within $1.5$IQR)};

      \end{tikzpicture}
      \end{center}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Components}
        \small
        \begin{itemize}
          \item Box: $Q1$ to $Q3$ (the IQR)
          \item Line inside box: median
          \item Whiskers: extend to most extreme point within $1.5\,\text{IQR}$
          \item Dots: outliers beyond the whiskers
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Using Summaries Sensibly}{Which number to report, and how}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Mean vs.\ median}
        \small
        \begin{itemize}
          \item Symmetric, no outliers $\Rightarrow$ mean is fine
          \item Skewed or outlier-prone $\Rightarrow$ prefer median
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{Reporting precision}
        \small
        \begin{itemize}
          \item Don't report more digits than the data supports
          \item Categorical data: report \keyword{mode}, not mean
          \item Always pair a \keyword{location} with a \keyword{spread}
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Case Study: Australian Pizzas}

\begin{frame}{Case Study — Whose Pizza Is Bigger?}{EagleBoys claims a bigger pizza than Dominos}

        \small An advertising claim: EagleBoys' pizzas are bigger than

      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{The dataset}

\begin{frame}{Conclusions}{Evidence, caveats, and the answer}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{Evidence for EagleBoys' claim}
        \small
        \begin{itemize}
          \item Higher mean \emph{and} median diameter
          \item Distributions barely overlap
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{Important caveats}
        \small
        \begin{itemize}
          \item Small sample — is the difference \keyword{significant}?
          \item Sampling bias: which stores, which day?
          \item Bigger diameter $\neq$ more \emph{value} (price, toppings)
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vfill
  \begin{block}{The answer}
    \small On this sample, EagleBoys' pizzas \emph{are} measurably larger —
    but a rigorous claim needs the hypothesis-testing tools of Part III.
  \end{block}
\end{frame}

\section{Summary}
% ============================================================

\begin{frame}{Location vs.\ Scale Parameters}{The two questions every summary answers}
  \small
  \renewcommand{\arraystretch}{1.3}
  \begin{center}
  \begin{tabularx}{\textwidth}{@{}>{\bfseries}l l >{\raggedright\arraybackslash}X@{}}
    \toprule
    \rowcolor{PrimaryDark}
    \textcolor{white}{Question} & \textcolor{white}{Statistic} & \textcolor{white}{Robust to outliers?} \\
    \midrule
    Where is the data? (location) & Mean & No \\
    \rowcolor{PrimaryLight!25}
     & Median & Yes \\
    How spread out? (scale) & Std.\ dev.\ $\sigma$ & No \\
    \rowcolor{PrimaryLight!25}
     & IQR & Yes \\
    \bottomrule
  \end{tabularx}
  \end{center}
\end{frame}

\begin{frame}{Chapter Summary}{Key takeaways}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Describe}
        \small
        \begin{itemize}
          \item Data types drive the choice of summary and plot
          \item Bar charts for categories, histograms for numbers
        \end{itemize}
      \end{block}
      \begin{exampleblock}{Summarize}
        \small Mean/median for location, std.\ dev./IQR for spread,
        z-scores to compare across scales.
      \end{exampleblock}
    \end{column}

      \begin{alertblock}{Watch out for}
        \small Outliers can distort the mean and variance — always pair a
        numeric summary with a \keyword{plot}.
      \end{alertblock}
