\section{The Correlation Coefficient}
% ============================================================

\begin{frame}{Defining Correlation}{Quantifying a linear relationship}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Definition}
        \small For standardized data $\{(\hat x_i,\hat y_i)\}$:
        \[ r = \text{corr}(\{(x,y)\}) = \frac{1}{N}\sum_{i=1}^N \hat x_i \hat y_i \]
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{Properties}
        \small
        \begin{itemize}
          \item $-1 \le r \le 1$
          \item Symmetric: $\text{corr}(x,y) = \text{corr}(y,x)$
          \item Unaffected by translation or positive scaling
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Interpreting the Sign and Magnitude}{Direction, then strength}
  \small
  \renewcommand{\arraystretch}{1.3}
  \begin{center}
  \begin{tabularx}{\textwidth}{@{}c >{\raggedright\arraybackslash}X@{}}
    \toprule
    \rowcolor{PrimaryDark}
    \textcolor{white}{Value of $r$} & \textcolor{white}{Interpretation} \\
    \midrule
    $r = +1$ & Perfect positive relationship ($\hat y = \hat x$) \\
    \rowcolor{PrimaryLight!25}
    $|r| \ge 0.7$ & Strong correlation \\
    $0.3 \le |r| < 0.7$ & Moderate correlation \\
    \rowcolor{PrimaryLight!25}
    $|r| < 0.3$ & Weak correlation \\
    $r = 0$ & No \emph{linear} relationship \\
    \rowcolor{PrimaryLight!25}
    $r = -1$ & Perfect negative relationship ($\hat y = -\hat x$) \\
    \bottomrule
  \end{tabularx}
  \end{center}
  \begin{alertblock}{These are guidelines}
    \small Interpretation always depends on the domain — $r=0.3$ can be
    remarkable in social science, unremarkable in physics.
  \end{alertblock}
\end{frame}

\section{Predicting from Correlation}
% ============================================================

\begin{frame}{The Best Linear Predictor}{Deriving the least-squares line in standard coordinates}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small
      Predict $\hat y_i^p = a\hat x_i + b$. Requiring $\text{mean}(u)=0$ for
      error $u=\hat y - \hat y^p$ forces $b=0$. Minimizing
      $\text{var}(u) = 1 - 2ar + a^2$ over $a$ gives:
      \[ \frac{d}{da}(1-2ar+a^2) = -2r+2a = 0 \;\Rightarrow\; a = r \]
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{Result}
        \small In standard coordinates:
        \[ \hat y^p = r\,\hat x \]
        \small In original coordinates:
        \[ y^p = \bar y + r\,\frac{\sigma_y}{\sigma_x}(x-\bar x) \]
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{How Good Is the Prediction?}{Error shrinks as $|r|$ grows}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{RMSE of the prediction}
        \[ \text{RMSE} = \sigma_y\sqrt{1-r^2} \]
      \end{block}
      \begin{exampleblock}{Rule of thumb}
        \small If $x$ is $k$ std.\ devs.\ from its mean, $y$ is predicted to
        be $r\cdot k$ std.\ devs.\ from its mean.
      \end{exampleblock}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \begin{tikzpicture}
        \begin{axis}[width=6.2cm, height=4.2cm, xlabel={\scriptsize $|r|$},
          ylabel={\scriptsize RMSE $/\ \sigma_y$}, tick label style={font=\scriptsize},
          axis lines=left, domain=0:1, samples=40, ymin=0]
          \addplot[PrimaryMid, thick] {sqrt(1-x^2)};
        \end{axis}
      \end{tikzpicture}\\
      \scriptsize Perfect correlation ($|r|{=}1$) $\Rightarrow$ zero error
    \end{column}
  \end{columns}
\end{frame}

\section{Correlation Is Not Causation}
% ============================================================

\begin{frame}{Spurious Correlation}{Ice cream sales and drowning deaths}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{The naive observation}
        \small Ice cream sales and drowning deaths rise and fall together
        across the year, $r$ is large and positive.
      \end{alertblock}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{The real explanation}
        \small A \keyword{lurking variable} — hot summer weather — drives
        both \emph{more swimming} and \emph{more ice cream}.
      \end{exampleblock}
    \end{column}
  \end{columns}
  \vfill
  \begin{center}
  \begin{tikzpicture}[node distance=1.1cm and 1.6cm,
      n/.style={rounded corners=2pt, draw=PrimaryMid, fill=PrimaryLight!40, font=\scriptsize, align=center, minimum height=0.6cm}]
    \node[n, fill=PrimaryDark, text=white] (z) {Hot weather};
    \node[n, below left=of z] (x) {Ice cream sales};
    \node[n, below right=of z] (y) {Drowning deaths};
    \draw[-{Latex[length=2mm]}, PrimaryMid, thick] (z) -- (x);
    \draw[-{Latex[length=2mm]}, PrimaryMid, thick] (z) -- (y);
    \draw[<->, AlertRed, dashed, thick] (x) -- node[below,font=\scriptsize]{spurious $r$} (y);
  \end{tikzpicture}
  \end{center}
\end{frame}

\begin{frame}{Confounding Variables}{Shoe size and reading ability}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{The naive observation}
        \small Across all schoolchildren, shoe size correlates strongly with
        reading test scores.
      \end{alertblock}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{The confound: age}
        \small Older children have \emph{both} bigger feet and better
        reading skills. Age is the true driver of both.
      \end{exampleblock}
    \end{column}
  \end{columns}
  \vfill
  \begin{block}{Three explanations for any observed $r$}
    \small
    \begin{enumerate}
      \item $X$ causes $Y$
      \item $Y$ causes $X$
      \item A hidden variable $Z$ causes both $X$ and $Y$
    \end{enumerate}
  \end{block}
\end{frame}

\section{Case Study: Wild Horse Herds}
% ============================================================

\begin{frame}{Case Study — Sterile Males in Wild Horse Herds}{A correlation that reversed sign}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{The problem}
        \small Does introducing sterile males reduce foal births? A naive
        correlation between \% sterile males and births per year was
        computed \emph{across all herds and years}.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{Surprising result}
        \small The naive correlation came out \important{positive} — more
        sterile males, \emph{more} births?
      \end{alertblock}
    \end{column}
  \end{columns}
  \vfill
  \begin{exampleblock}{The key insight: time as a lurking variable}
    \small Herd size and the sterile-male program both grew over
    \keyword{time}. Correlating \emph{within} each year (controlling for
    time) reveals the expected \important{negative} relationship.
  \end{exampleblock}
\end{frame}

\section{Pitfalls \& Alternatives}
% ============================================================

\begin{frame}{Three Pitfalls of Correlation}{Where $r$ can badly mislead you}
  \small
  \renewcommand{\arraystretch}{1.3}
  \begin{center}
  \begin{tabularx}{\textwidth}{@{}>{\bfseries}l >{\raggedright\arraybackslash}X@{}}
    \toprule
    \rowcolor{PrimaryDark}
    \textcolor{white}{Pitfall} & \textcolor{white}{What goes wrong} \\
    \midrule
    Non-linear relationships & $r\approx 0$ even for a strong curved pattern \\
    \rowcolor{PrimaryLight!25}
    Outliers & A single point can swing $r$ dramatically \\
    Scaling / grouping & Combining subgroups can flip the sign (Simpson-like effect) \\
    \bottomrule
  \end{tabularx}
  \end{center}
  \begin{alertblock}{Golden rule}
    \small Always plot the scatter plot before trusting a single number.
  \end{alertblock}
\end{frame}

\begin{frame}{Spearman's Rank Correlation}{A robust alternative to Pearson's $r$}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Idea}
        \small Replace each value by its \keyword{rank}, then compute
        Pearson correlation on the ranks.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{Why it's robust}
        \small
        \begin{itemize}
          \item Detects any \keyword{monotonic} relationship, not just linear
          \item Outliers only shift a rank by one place
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Chapter Summary}{Key takeaways}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Visualize}
        \small Scatter plots first, always — correlation is only a number.
      \end{block}
      \begin{exampleblock}{Quantify \& predict}
        \small $r$ measures linear association; the best linear predictor
        is $y^p = \bar y + r\frac{\sigma_y}{\sigma_x}(x-\bar x)$.
      \end{exampleblock}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{Never forget}
        \small Correlation $\neq$ causation — always ask about \keyword{lurking}
        and \keyword{confounding} variables.
      \end{alertblock}
